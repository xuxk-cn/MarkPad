```rust
use std::sync::Mutex;
use ropey::Rope;

/// 全局文档状态（单一事实源）
pub struct DocumentState {
    pub rope: Rope,
}

impl DocumentState {
    pub fn new() -> Self {
        Self {
            rope: Rope::new(),
        }
    }

    pub fn insert(&mut self, offset: usize, text: &str) {
        if offset <= self.rope.len_chars() {
            self.rope.insert(offset, text);
        }
    }

    pub fn delete(&mut self, start: usize, end: usize) {
        if end <= self.rope.len_chars() && start < end {
            self.rope.remove(start..end);
        }
    }

    pub fn get_all_text(&self) -> String {
        self.rope.to_string()
    }

    pub fn len(&self) -> usize {
        self.rope.len_chars()
    }
}

// 放入 Tauri 状态管理
pub type Document = Mutex<DocumentState>;
```

```typescript
/**
 * 极简 Markdown → HTML 内联渲染（仅支持 MVP 语法）
 * Phase 0 只做基本块级识别，后续引入 pulldown-cmark WASM
 */
export function renderMarkdown(md: string): string {
  const lines = md.split("\n");
  return lines.map(line => {
    if (line.startsWith("# ")) return `<h1>${escapeHtml(line.slice(2))}</h1>`;
    if (line.startsWith("## ")) return `<h2>${escapeHtml(line.slice(3))}</h2>`;
    if (line.startsWith("### ")) return `<h3>${escapeHtml(line.slice(4))}</h3>`;
    if (line.startsWith("- ")) return `<li>${escapeHtml(line.slice(2))}</li>`;
    if (line.startsWith("> ")) return `<blockquote>${escapeHtml(line.slice(2))}</blockquote>`;
    if (line.trim() === "") return "<br>";
    return `<p>${escapeHtml(line)}</p>`;
  }).join("");
}

function escapeHtml(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}
```

**Step 2: 定义块级 AST**

创建 `src-tauri/src/parser.rs`：

```rust
use pulldown_cmark::{Event, Parser, Tag, TagEnd};
use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct Block {
    /// 块起始字符偏移（在原始 Markdown 中的位置）
    pub start: usize,
    /// 块结束字符偏移
    pub end: usize,
    /// 块类型
    pub kind: BlockKind,
}

#[derive(Debug, Clone, Serialize)]
pub enum BlockKind {
    Heading { level: u8 },
    Paragraph,
    CodeBlock { lang: Option<String> },
    BlockQuote,
    ListItem { ordered: bool },
    ThematicBreak,
    BlankLine,
}

/// 解析 Markdown 文本，返回所有块级元素
pub fn parse_blocks(md: &str) -> Vec<Block> {
    let parser = Parser::new(md);
    let mut blocks = Vec::new();

    for event in parser {
        match event {
            Event::Start(Tag::Heading { level, .. }) => {
                // pulldown-cmark 的 span 信息通过 offset 获取
                // 简化版：用正则预扫描定位。Phase 1 会改进。
            }
            _ => {}
        }
    }

    blocks
}
```

注意：`pulldown-cmark` 需要启用 `html` feature 来获取 `TagEnd`。

**Step 3: 完整实现（使用 pulldown-cmark 的 Parser::new_with_broken_link_callback 的 offset 功能）**

pulldown-cmark 0.12 的 `Parser::into_offset_iter()` 提供了每个 Event 的字节偏移。利用这个精确定位块边界。

```rust
use pulldown_cmark::{Event, Options, Parser, Tag, TagEnd};

pub fn parse_blocks(md: &str) -> Vec<Block> {
    let mut options = Options::all();
    let parser = Parser::new_ext(md, options).into_offset_iter();
    let mut blocks = Vec::new();

    for (event, range) in parser {
        match event {
            Event::Start(Tag::Heading { level, .. }) => {
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::Heading { level: level as u8 },
                });
            }
            Event::Start(Tag::Paragraph) => {
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::Paragraph,
                });
            }
            Event::Start(Tag::CodeBlock(kind)) => {
                let lang = match kind {
                    pulldown_cmark::CodeBlockKind::Fenced(s) => {
                        if s.is_empty() { None } else { Some(s.to_string()) }
                    }
                    _ => None,
                };
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::CodeBlock { lang },
                });
            }
            Event::Start(Tag::BlockQuote(_)) => {
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::BlockQuote,
                });
            }
            Event::Start(Tag::List(_)) | Event::Start(Tag::Item) => {
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::ListItem { ordered: false },
                });
            }
            Event::Rule => {
                blocks.push(Block {
                    start: range.start,
                    end: range.end,
                    kind: BlockKind::ThematicBreak,
                });
            }
            _ => {}
        }
    }

    blocks
}
```

```typescript
/** Web Worker: 接收 Markdown 文本，返回块级 AST（简化版，后续调用 WASM） */
import { renderMarkdown } from "../renderer";

self.onmessage = (e: MessageEvent<{ md: string; viewStart: number; viewEnd: number }>) => {
  const { md, viewStart, viewEnd } = e.data;
  const lines = md.split("\n");

  // 只处理视口内的行（± 缓冲）
  const buffer = 20; // 上下各 20 行
  const start = Math.max(0, viewStart - buffer);
  const end = Math.min(lines.length, viewEnd + buffer);

  const blocks: { lineIndex: number; html: string }[] = [];
  for (let i = start; i < end; i++) {
    blocks.push({
      lineIndex: i,
      html: renderMarkdown(lines[i]),
    });
  }

  self.postMessage({ blocks, totalLines: lines.length });
};
```

**Step 2: 视口检测**

`src/workers/render-scheduler.ts`：

```typescript
/**
 * 视口裁剪调度器
 * 监听滚动事件，只渲染可见区域的块
 */
export class ViewportScheduler {
  private scrollContainer: HTMLElement;
  private lineHeight: number;
  private renderCallback: (start: number, end: number) => void;
  private ticking = false;

  constructor(
    scrollContainer: HTMLElement,
    lineHeight: number,
    renderCallback: (start: number, end: number) => void,
  ) {
    this.scrollContainer = scrollContainer;
    this.lineHeight = lineHeight;
    this.renderCallback = renderCallback;
  }

  /** 启动视口监听 */
  start(): void {
    this.scrollContainer.addEventListener("scroll", () => this.scheduleUpdate());
    // 初始渲染
    this.scheduleUpdate();
  }

  private scheduleUpdate(): void {
    if (!this.ticking) {
      this.ticking = true;
      requestAnimationFrame(() => {
        this.update();
        this.ticking = false;
      });
    }
  }

  private update(): void {
    const scrollTop = this.scrollContainer.scrollTop;
    const clientHeight = this.scrollContainer.clientHeight;

    const startLine = Math.floor(scrollTop / this.lineHeight);
    const endLine = Math.ceil((scrollTop + clientHeight) / this.lineHeight);

    this.renderCallback(startLine, endLine);
  }
}
```

**Step 3: 集成到 editor**

在 `src/editor.ts` 中，输入时：
1. 调用 `invoke("parse_document")` 获取块列表（后续改为 Rust→前端推送）
2. 使用 `ViewportScheduler` 控制渲染范围

```typescript
export function setupIncrementalRender(
  sourceEl: HTMLElement,
  overlayEl: HTMLElement,
): void {
  const scheduler = new ViewportScheduler(
    sourceEl,
    24, // line-height in px
    (start, end) => {
      // 调用 Worker 渲染
      const text = sourceEl.innerText ?? "";
      const worker = new Worker(
        new URL("./workers/parser.worker.ts", import.meta.url),
        { type: "module" },
      );
      worker.postMessage({ md: text, viewStart: start, viewEnd: end });
      worker.onmessage = (e) => {
        const { blocks } = e.data;
        updateOverlayBlocks(overlayEl, blocks, start);
      };
    },
  );
  scheduler.start();
}

function updateOverlayBlocks(
  overlayEl: HTMLElement,
  blocks: { lineIndex: number; html: string }[],
  offset: number,
): void {
  // 只更新 overlay 中变动的行，不做全量 innerHTML
  const lines = overlayEl.children;
  for (const block of blocks) {
    const idx = block.lineIndex;
    if (idx < lines.length) {
      lines[idx].innerHTML = block.html;
    }
  }
}
```

```rust
pub fn generate_fixture(lines: usize) -> String {
    let mut md = String::with_capacity(lines * 50);
    for i in 0..lines {
        match i % 10 {
            0 => md.push_str(&format!("# Section {}\n\n", i / 10)),
            1 => md.push_str(&format!("## Subsection {}.{}\n\n", i / 10, i % 10)),
            2 => md.push_str(&format!("This is paragraph {}. It contains some **bold** and *italic* text, and a [link](https://example.com).\n\n", i)),
            3 => md.push_str(&format!("- Item {} in a list\n- Another item\n- Third item\n\n", i)),
            4 => md.push_str(&format!("> This is a blockquote line {}.\n> Multiple lines of quoted text.\n\n", i)),
            5 => md.push_str(&format!("```rust\nfn hello_{}() {{\n    println!(\"Hello, world!\");\n}}\n```\n\n", i)),
            6 => md.push_str("\n"), // blank line
            7 => md.push_str(&format!("### Sub-subsection {}\n\n", i)),
            8 => md.push_str(&format!("Regular paragraph {} with `inline code` and more text to fill the line.\n\n", i)),
```
